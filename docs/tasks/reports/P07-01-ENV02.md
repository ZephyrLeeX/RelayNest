```js
const { chromium } = await import(process.env.PLAYWRIGHT_MODULE);

try {
  const context = await browser.newContext();
  const page = await context.newPage();
  await page.goto('data:text/html;charset=utf-8,' + encodeURIComponent('<!doctype html><html><body style="font-family:DejaVu Sans"><p id="text">RelayNest environment font smoke 123</p><button onclick="document.body.dataset.clicked=String(6*7)">Run JS</button></body></html>'));
  await page.evaluate(() => document.fonts.ready);
  await page.getByRole('button', { name: 'Run JS' }).click();
  const result = await page.evaluate(() => ({ js: document.body.dataset.clicked, font: getComputedStyle(document.querySelector('#text')).fontFamily, fontReady: document.fonts.check('16px "DejaVu Sans"'), width: document.querySelector('#text').getBoundingClientRect().width }));
  assert.equal(result.js, '42'); assert.equal(result.fontReady, true); assert(result.width > 0);
  const cdp = await context.newCDPSession(page);
  await cdp.send('DOM.enable'); await cdp.send('CSS.enable');
  const { root } = await cdp.send('DOM.getDocument');
  const { nodeId } = await cdp.send('DOM.querySelector', { nodeId: root.nodeId, selector: '#text' });
  const { fonts } = await cdp.send('CSS.getPlatformFontsForNode', { nodeId });
  assert(fonts.some(font => font.postScriptName === 'DejaVuSans' && font.glyphCount > 0));
  console.log(JSON.stringify({ result: 'PASS', browser: browser.version(), ...result, renderedFonts: fonts }));
  await context.close();
} finally { await browser.close(); }
```
